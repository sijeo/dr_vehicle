 1403: /* =====================================================================
 1404:  * Production IMU calibration framework (imu_calib_t).
 1405:  *
 1406:  * Frame conventions:
 1407:  *   IMU physical mounting (sensor frame): FRD, right-handed
 1408:  *     X = sensor X — forward (along chassis)
 1409:  *     Y = sensor Y — vehicle right        ← updated per hardware install
 1410:  *     Z = sensor Z — down (toward the road)
 1411:  *     A right-handed sensor chip mounted with X→front and Y→right has
 1412:  *     its Z axis pointing down by construction (X×Y = Z = forward×right
 1413:  *     = down); the chip reads ~-g on Z when stationary.
 1414:  *
 1415:  *   Math vehicle frame (used by AHRS, INS, EKF, ENU integration): FLU
 1416:  *     X_v = vehicle forward
 1417:  *     Y_v = vehicle left
 1418:  *     Z_v = vehicle up
 1419:  *     Gravity reads as +g on Z_v (the math expects this).
 1420:  *
 1421:  * mount_R_bv rotates a vector from the IMU sensor frame into the math
 1422:  * vehicle frame:  v_vehicle = mount_R_bv * v_imu_body.
 1423:  *
 1424:  * Default mount_R_bv = diag(1, -1, -1) — a 180° rotation about the X
 1425:  * axis that converts FRD sensor → FLU math:
 1426:  *   vehicle_X_forward =  imu_X_forward    (unchanged)
 1427:  *   vehicle_Y_left    = -imu_Y_right      (Y flips)
 1428:  *   vehicle_Z_up      = -imu_Z_down       (Z flips)
 1429:  *
 1430:  * Why not change the math to FRU? FRU (forward-right-up) is left-handed
 1431:  * (forward × right = down ≠ up), and every rotation, quaternion, cross
 1432:  * product, and ENU↔body conversion in this codebase assumes right-handed
 1433:  * conventions. The mount_R_bv default lets the IMU be installed in the
 1434:  * preferred FRD orientation (X→front, Y→right) without touching the
 1435:  * EKF / AHRS internals.
 1436:  *
 1437:  * If the IMU is ever installed in a different orientation, override
 1438:  * mount_R_bv via the persisted calibration blob; boot calibration
 1439:  * preserves the existing mount_R_bv (see apply_poweron_calibration).
 1440:  * ===================================================================== */

 1485: static uint32_t crc32_simple(const void *data, size_t nbytes) {
 1486:     size_t i;
 1487:     int b;
 1488:     const uint8_t *p = (const uint8_t*)data;
 1489:     uint32_t crc = 0xFFFFFFFFu;
 1490:     for (i = 0; i < nbytes; i++) {
 1491:         crc ^= p[i];
 1492:         for (b = 0; b < 8; b++) {
 1493:             uint32_t m = (uint32_t)-(int32_t)(crc & 1u);

 1496:     }

 1552:     cal->mount_R_bv[1][0] =  0.0f; cal->mount_R_bv[1][1] = -1.0f; cal->mount_R_bv[1][2] =  0.0f;
 1553:     cal->mount_R_bv[2][0] =  0.0f; cal->mount_R_bv[2][1] =  0.0f; cal->mount_R_bv[2][2] = -1.0f;
 1554: 
 1555:     cal->temperature_c = 25.0f;
 1556:     cal->valid_flags   = 0;
 1557: }

 1559: /* Sanity validation — returns 0 if ok, negative reason code otherwise.
 1560:  *   -1 = magic/version mismatch
 1561:  *   -2 = checksum mismatch (caller should compute before)
 1562:  *   -3 = gyro bias out of range
 1563:  *   -4 = accel bias out of range
 1564:  *   -5 = accel norm mean out of plausible gravity range
 1565:  *   -6 = variance excessive (vibration / motion during cal)
 1566:  *   -7 = NaN/Inf in any field
 1567:  */
 1568: static int imu_calib_validate(const imu_calib_t *cal) {
 1569:     int i, j;
 1570:     if (cal->magic != IMU_CALIB_MAGIC)   return -1;
 1571:     if (cal->version != IMU_CALIB_VERSION) return -1;
 1572:     // NaN/Inf scan over float members
 1573:     const float *f = (const float *)&cal->accel_bias[0];
 1574:     size_t nf = (offsetof(imu_calib_t, temperature_c) +
 1575:                  sizeof(cal->temperature_c) -
 1576:                  offsetof(imu_calib_t, accel_bias)) / sizeof(float);
 1577:     for (i = 0; i < (int)nf; i++) {
 1578:         float v = f[i];
 1579:         if (!(v == v) || (v > 1e30f) || (v < -1e30f)) return -7;
 1580:     }
 1581:     for (i = 0; i < 3; i++) {
 1582:         if (fabsf(cal->gyro_bias[i])  > CAL_SANITY_GYRO_BIAS_MAX)  return -3;
 1583:         if (fabsf(cal->accel_bias[i]) > CAL_SANITY_ACCEL_BIAS_MAX) return -4;
 1584:         // Scales must be near 1
 1585:         if (cal->accel_scale[i] < 0.5f || cal->accel_scale[i] > 1.5f) return -4;
 1586:         if (cal->gyro_scale [i] < 0.5f || cal->gyro_scale [i] > 1.5f) return -3;
 1587:     }
 1588:     if (cal->boot_sample_count > 0) {
 1589:         if (cal->boot_accel_norm_mean < CAL_SANITY_ACCEL_NORM_MIN ||
 1590:             cal->boot_accel_norm_mean > CAL_SANITY_ACCEL_NORM_MAX) return -5;
 1591:         for (i = 0; i < 3; i++) {
 1592:             if (cal->boot_accel_var[i] > CAL_SANITY_ACC_VAR_MAX)  return -6;
 1593:             if (cal->boot_gyro_var [i] > CAL_SANITY_GYRO_VAR_MAX) return -6;
 1594:         }
 1595:     }
 1596:     // Mount/misalign rows finite and not absurd
 1597:     for (i = 0; i < 3; i++) for (j = 0; j < 3; j++) {
 1598:         float m = cal->mount_R_bv[i][j];
 1599:         if (!(m == m) || fabsf(m) > 2.0f) return -7;
 1600:         float a = cal->accel_misalign[i][j];
 1601:         if (!(a == a) || fabsf(a) > 2.0f) return -7;
 1602:     }
 1603:     return 0;
 1604: }

 1645: /* Bridge: convert legacy g_cal (LSQ + raw gyro counts) into the production
 1646:  * imu_calib_t bias fields. The 6-position LSQ accelerometer matrix in g_cal
 1647:  * already corrects for scale and cross-axis errors, so accel_misalign/scale
 1648:  * here stay at identity and the residual offset is folded into accel_bias.
 1649:  * Called whenever g_cal changes (boot cal completes or a fresh default load). */
 1650: static void imu_calib_sync_from_g_cal(imu_calib_t *cal) {
 1651:     int i;
 1652:     const float L = DEG2RAD / GYRO_LSB_PER_DPS;
 1653:     for (i = 0; i < 3; i++) {
 1654:         cal->gyro_bias[i]  = g_cal.gyro_bias_counts[i] * L;
 1655:         cal->accel_bias[i] = 0.0f;   // accel offset is baked into g_cal.accel_O
 1656:     }
 1657:     cal->gyro_scale[2] *= g_cal.gyro_z_scale;
 1658: }

 1660: static void cal_set_defaults_from_lsq( void ) {
 1661:     memset(&g_cal, 0, sizeof(g_cal));
 1662:     g_cal.magic = CAL_MAGIC;
 1663:     g_cal.version = CAL_VERSION;
 1664: 
 1665:     /* ---------------------------------------------------------------------
 1666:      * LSQ accelerometer calibration coefficients — ISM330DLC at ±4g FS.
 1667:      *
 1668:      * Calibration model:   a_mps2 = accel_C · raw_counts + accel_O
 1669:      *
 1670:      * The 3×3 matrix encodes per-axis scale, cross-axis correction, AND
 1671:      * the physical PCB-mount permutation captured during the 6-position
 1672:      * fit (note that the dominant coefficient on row 0 is column 2: the
 1673:      * IMU's chip-Z axis lands on the vehicle-X output, etc.).
 1674:      *
 1675:      * Re-scale from the original ±2g fit to ±4g:
 1676:      *   ±2g sensitivity: 16384 LSB/g  → 1 LSB ≈ 5.99e-4 m/s²
 1677:      *   ±4g sensitivity:  8192 LSB/g  → 1 LSB ≈ 1.20e-3 m/s² (2×)
 1678:      *
 1679:      * For the SAME physical acceleration, the ±4g ADC produces exactly
 1680:      * half the counts of the ±2g ADC. To preserve a_mps2 we must have:
 1681:      *     C_4g = 2 · C_2g           (each LSB now represents 2× m/s²)
 1682:      *     O_4g = O_2g               (because C·counts_at_zero is preserved
 1683:      *                                when both C doubles and counts halve)
 1684:      *
 1685:      * This is an exact rescale of the calibration, not a degraded
 1686:      * approximation — the LSQ residuals are identical to the original
 1687:      * ±2g fit. If the IMU full-scale range is ever changed back to ±2g
 1688:      * (or to ±8g / ±16g), recompute by the rule:
 1689:      *     C_new = C_2g · (FS_new / 2g)
 1690:      *     O_new = O_2g                                                  */
 1691:     g_cal.accel_C[0][0] = -4.5500962626366430e-05f; g_cal.accel_C[0][1] =  1.1530531391926312e-05f; g_cal.accel_C[0][2] = 0.0011976699942999136f;
 1692:     g_cal.accel_C[1][0] = -4.7019713060111626e-05f; g_cal.accel_C[1][1] =  0.0012001534090637276f; g_cal.accel_C[1][2] = -6.7647153361837210e-06f;
 1693:     g_cal.accel_C[2][0] =  0.0011835791325487338f;  g_cal.accel_C[2][1] =  1.1158793472464106e-04f; g_cal.accel_C[2][2] =  1.0592350070188138e-05f;
 1694: 
 1695:     /* Offset stays unchanged across an FS-range rescale (see derivation above). */
 1696:     g_cal.accel_O[0] = -0.04324381676101664f;
 1697:     g_cal.accel_O[1] =  0.16600572009786152f;
 1698:     g_cal.accel_O[2] =  0.5714660018044386f;
 1699: 
 1700:     /* Gyro defaults are independent of the accel range. ISM330DLC gyro
 1701:      * is configured at ±500 dps (see GYRO_LSB_PER_DPS) — these bias
 1702:      * counts come from the original power-on stationary capture and
 1703:      * get overwritten by boot calibration anyway. */
 1704:     g_cal.gyro_bias_counts[0] = -38.741f;
 1705:     g_cal.gyro_bias_counts[1] =  57.638f;
 1706:     g_cal.gyro_bias_counts[2] = -16.246f;
 1707: 
 1708:     g_cal.gyro_z_scale = 1.0f;
 1709:     g_cal.accel_x_scale = 1.0f;
 1710:     g_cal.calibrated_once = 0;
 1711: }

 1775: static void calib_accel(const struct mpu6050_sample *raw, vec3f *acc_mps2) {
 1776:     float a0 = (float)raw->ax, a1 = (float)raw->ay, a2 = (float)raw->az;

 1779:     float y = g_cal.accel_C[1][0]*a0 + g_cal.accel_C[1][1]*a1 + g_cal.accel_C[1][2]*a2 + g_cal.accel_O[1];
 1780:     float z = g_cal.accel_C[2][0]*a0 + g_cal.accel_C[2][1]*a1 + g_cal.accel_C[2][2]*a2 + g_cal.accel_O[2];

 1790: static void calib_gyro(const struct mpu6050_sample *raw, vec3f *gyro_radps) {
 1791:     float g0 = (float)raw->gx - g_cal.gyro_bias_counts[0];
 1792:     float g1 = (float)raw->gy - g_cal.gyro_bias_counts[1];
 1793:     float g2 = (float)raw->gz - g_cal.gyro_bias_counts[2];
 1794:     
 1795:     float dps0 = g0 / GYRO_LSB_PER_DPS;
 1796:     float dps1 = g1 / GYRO_LSB_PER_DPS;

 1804: /* =====================================================================
 1805:  * Centralized IMU preprocessing
 1806:  *
 1807:  * Source-of-truth convention for biases (avoid double-subtract):
 1808:  *   g_cal              — owns the APPLIED bias / LSQ matrix / scale used
 1809:  *                        by calib_accel() and calib_gyro().
 1810:  *   g_imu_cal.*_bias   — mirrors the same SI bias values for persistence
 1811:  *                        and diagnostics ONLY.
 1812:  * imu_preprocess_sample() therefore does NOT re-subtract g_imu_cal.*_bias
 1813:  * (it would be applied twice). It only adds the corrections that g_cal
 1814:  * cannot represent: per-axis residual scale, cross-axis misalignment and
 1815:  * the mount→vehicle rotation.
 1816:  *
 1817:  * Order of operations:
 1818:  *   1. raw counts → physical units            (calib_accel / calib_gyro)
 1819:  *   2. accel_misalign (3×3, default identity)
 1820:  *   3. per-axis residual scale (gyro_scale, accel_scale)
 1821:  *   4. mount_R_bv: rotate IMU/body frame into vehicle frame
 1822:  *
 1823:  * The output vectors are in vehicle frame and SI units:
 1824:  *   accel_vehicle[] = m/s² with gravity still present
 1825:  *   gyro_vehicle[]  = rad/s
 1826:  *
 1827:  * Median/low-pass filtering is applied separately in the fusion thread
 1828:  * after this function returns, because the filter state is shared with
 1829:  * the bump detector. ===================================================*/
 1830: static void imu_preprocess_sample(const imu_calib_t *cal,
 1831:                                   const struct mpu6050_sample *raw,
 1832:                                   float accel_vehicle[3],
 1833:                                   float gyro_vehicle[3])
 1834: {
 1835:     /* 1) Raw → SI using legacy g_cal (bias already removed inside) */
 1836:     vec3f acc_b, gyro_b;
 1837:     calib_accel(raw,  &acc_b);
 1838:     calib_gyro (raw,  &gyro_b);
 1839: 
 1840:     /* 2) Optional accel cross-axis (misalignment) correction */
 1841:     float a_in[3]  = { acc_b.x,  acc_b.y,  acc_b.z };
 1842:     float a_aligned[3];
 1843:     mat3_mul_vec(cal->accel_misalign, a_in, a_aligned);
 1844: 
 1845:     /* 3) Per-axis residual scale (default 1.0 — no-op until set by runtime) */
 1846:     a_aligned[0] *= cal->accel_scale[0];
 1847:     a_aligned[1] *= cal->accel_scale[1];
 1848:     a_aligned[2] *= cal->accel_scale[2];
 1849: 
 1850:     float g_in[3] = { gyro_b.x, gyro_b.y, gyro_b.z };
 1851:     g_in[0] *= cal->gyro_scale[0];
 1852:     g_in[1] *= cal->gyro_scale[1];
 1853:     g_in[2] *= cal->gyro_scale[2];
 1854: 
 1855:     /* 4) IMU/body → vehicle frame (default mount = identity) */
 1856:     apply_mount_rotation(cal, a_aligned, accel_vehicle);
 1857:     apply_mount_rotation(cal, g_in,      gyro_vehicle);
 1858: }

 1860: /* =====================================================================
 1861:  * Conservative runtime gyro-bias learner.
 1862:  *
 1863:  * Updates g_imu_cal.gyro_bias (and the legacy g_cal counts) with a slow
 1864:  * LPF only when the vehicle has been confirmed stationary for at least
 1865:  * RT_CAL_WINDOW_S seconds. Accelerometer bias is intentionally NOT
 1866:  * updated at runtime — that requires confident attitude.
 1867:  *
 1868:  * Save is throttled to RT_CAL_MIN_SAVE_INTERVAL_S between persistence
 1869:  * writes; the file write is also performed on clean program exit.
 1870:  *
 1871:  * Returns true if a bias update was applied this call. =================*/
 1872: typedef struct {
 1873:     bool   armed;
 1874:     double t_start;
 1875:     double sum[3];        // rad/s
 1876:     double sum_sq[3];     // rad²/s²
 1877:     int    N;
 1878: } rt_cal_t;
 1879: 
 1880: static rt_cal_t g_rt_cal;
 1881: 
 1882: #if ENABLE_RUNTIME_GYRO_CAL
 1883: static bool runtime_gyro_cal_update(const float gyro_vehicle_rps[3],
 1884:                                     const float accel_vehicle_mps2[3],
 1885:                                     bool ekf_speed_low,
 1886:                                     bool gnss_speed_low,
 1887:                                     double now_s)
 1888: {
 1889:     float gnorm = sqrtf(gyro_vehicle_rps[0]*gyro_vehicle_rps[0] +
 1890:                         gyro_vehicle_rps[1]*gyro_vehicle_rps[1] +
 1891:                         gyro_vehicle_rps[2]*gyro_vehicle_rps[2]);
 1892:     float anorm = sqrtf(accel_vehicle_mps2[0]*accel_vehicle_mps2[0] +
 1893:                         accel_vehicle_mps2[1]*accel_vehicle_mps2[1] +
 1894:                         accel_vehicle_mps2[2]*accel_vehicle_mps2[2]);
 1895:     float adev  = fabsf(anorm - GRAVITY);
 1896:     bool still = (gnorm < (RT_CAL_GYRO_STILL_RAD * 1.7320508f)) &&  // ~root3 of per-axis
 1897:                  (adev  < RT_CAL_ACC_DEV_MPS2) &&
 1898:                  ekf_speed_low && gnss_speed_low &&
 1899:                  !g_imu_quality_degraded;
 1900: 
 1901:     if (!still) {
 1902:         g_rt_cal.armed = false;
 1903:         return false;
 1904:     }
 1905: 
 1906:     if (!g_rt_cal.armed) {
 1907:         g_rt_cal.armed   = true;
 1908:         g_rt_cal.t_start = now_s;
 1909:         g_rt_cal.N       = 0;
 1910:         g_rt_cal.sum[0] = g_rt_cal.sum[1] = g_rt_cal.sum[2] = 0.0;
 1911:         g_rt_cal.sum_sq[0] = g_rt_cal.sum_sq[1] = g_rt_cal.sum_sq[2] = 0.0;
 1912:     }
 1913:     g_rt_cal.sum[0]    += gyro_vehicle_rps[0];
 1914:     g_rt_cal.sum[1]    += gyro_vehicle_rps[1];
 1915:     g_rt_cal.sum[2]    += gyro_vehicle_rps[2];
 1916:     g_rt_cal.sum_sq[0] += (double)gyro_vehicle_rps[0] * gyro_vehicle_rps[0];
 1917:     g_rt_cal.sum_sq[1] += (double)gyro_vehicle_rps[1] * gyro_vehicle_rps[1];
 1918:     g_rt_cal.sum_sq[2] += (double)gyro_vehicle_rps[2] * gyro_vehicle_rps[2];
 1919:     g_rt_cal.N++;
 1920: 
 1921:     if ((now_s - g_rt_cal.t_start) < RT_CAL_WINDOW_S || g_rt_cal.N < 50)
 1922:         return false;
 1923: 
 1924:     /* Variance gate — verifies the window really was still */
 1925:     double invN = 1.0 / (double)g_rt_cal.N;
 1926:     double v0 = g_rt_cal.sum_sq[0]*invN - (g_rt_cal.sum[0]*invN)*(g_rt_cal.sum[0]*invN);
 1927:     double v1 = g_rt_cal.sum_sq[1]*invN - (g_rt_cal.sum[1]*invN)*(g_rt_cal.sum[1]*invN);
 1928:     double v2 = g_rt_cal.sum_sq[2]*invN - (g_rt_cal.sum[2]*invN)*(g_rt_cal.sum[2]*invN);
 1929:     if (v0 > CAL_SANITY_GYRO_VAR_MAX || v1 > CAL_SANITY_GYRO_VAR_MAX || v2 > CAL_SANITY_GYRO_VAR_MAX) {
 1930:         g_rt_cal.armed = false;
 1931:         return false;
 1932:     }
 1933: 
 1934:     float mean[3] = { (float)(g_rt_cal.sum[0]*invN),
 1935:                       (float)(g_rt_cal.sum[1]*invN),
 1936:                       (float)(g_rt_cal.sum[2]*invN) };
 1937: 
 1938:     /* The gyro samples passed into this function were produced by
 1939:      * calib_gyro(), which already subtracted g_cal.gyro_bias_counts.
 1940:      * The residual 'mean' is therefore the small drift not yet accounted
 1941:      * for. We push that residual directly into g_cal.gyro_bias_counts
 1942:      * (the single source of truth used by every preprocessing path),
 1943:      * scaled by RT_CAL_ALPHA so adaptation is slow. */
 1944:     float old_bias_rps[3] = {
 1945:         g_cal.gyro_bias_counts[0] * (DEG2RAD / GYRO_LSB_PER_DPS),
 1946:         g_cal.gyro_bias_counts[1] * (DEG2RAD / GYRO_LSB_PER_DPS),
 1947:         g_cal.gyro_bias_counts[2] * (DEG2RAD / GYRO_LSB_PER_DPS)
 1948:     };
 1949:     const float counts_per_rps = GYRO_LSB_PER_DPS / DEG2RAD;
 1950:     g_cal.gyro_bias_counts[0] += RT_CAL_ALPHA * mean[0] * counts_per_rps;
 1951:     g_cal.gyro_bias_counts[1] += RT_CAL_ALPHA * mean[1] * counts_per_rps;
 1952:     g_cal.gyro_bias_counts[2] += RT_CAL_ALPHA * mean[2] * counts_per_rps;
 1953:     float new_bias_rps[3] = {
 1954:         g_cal.gyro_bias_counts[0] * (DEG2RAD / GYRO_LSB_PER_DPS),

 1986: /* =====================================================================
 1987:  *   AHRS / complementary attitude filter
 1988:  *
 1989:  * Vehicle (= body) frame conventions in this file:
 1990:  *     X = vehicle forward
 1991:  *     Y = vehicle left
 1992:  *     Z = vehicle up
 1993:  *     +ωz (gyro Z) = counter-clockwise from above = left turn = +yaw rate
 1994:  *
 1995:  * Yaw convention (ENU, matching the rest of this codebase):
 1996:  *     yaw = 0     → vehicle X axis points East
 1997:  *     yaw = +π/2  → vehicle X axis points North
 1998:  *
 1999:  * State:
 2000:  *   roll, pitch, yaw — Euler angles, derived from quaternion q after each
 2001:  *                      update. q is the authoritative attitude (body→ENU).
 2002:  *
 2003:  * Update sequence on each IMU step (called from fusion_thread):
 2004:  *   1. gyro integration             → predict q
 2005:  *   2. accelerometer correction     → blend roll/pitch toward acc-derived
 2006:  *                                      angles, gated by ahrs_accel_is_trusted
 2007:  *   3. GNSS course correction       → blend yaw toward GNSS course, gated by
 2008:  *                                      ahrs_gnss_yaw_is_trusted
 2009:  *   4. extract Euler outputs        → roll/pitch/yaw used by ahrs_remove_gravity
 2010:  *                                      and by the 2D EKF as the yaw reference
 2011:  * ===================================================================== */

 2071: /* Quaternion → Euler (ZYX) extraction used by the AHRS.
 2072:  *   yaw   = atan2(2(wz + xy), 1 - 2(y² + z²))
 2073:  *   pitch = asin (2(wy - zx))            — clamped before asin
 2074:  *   roll  = atan2(2(wx + yz), 1 - 2(x² + y²))
 2075:  * Same convention as q_from_euler() defined earlier in this file. */
 2076: static void q_to_euler_zyx(quatf q, float *roll, float *pitch, float *yaw) {
 2077:     q = q_normalize(q);
 2078:     float w = q.w, x = q.x, y = q.y, z = q.z;
 2079:     float sinp = 2.0f * (w*y - z*x);
 2080:     if (sinp >  1.0f) sinp =  1.0f;
 2081:     if (sinp < -1.0f) sinp = -1.0f;
 2082:     *pitch = asinf(sinp);
 2083:     *roll  = atan2f(2.0f*(w*x + y*z), 1.0f - 2.0f*(x*x + y*y));
 2084:     *yaw   = atan2f(2.0f*(w*z + x*y), 1.0f - 2.0f*(y*y + z*z));
 2085: }

 2087: /* Initialize AHRS from a stationary accelerometer sample and (optional) GNSS yaw.
 2088:  *   accel_vehicle[]   — vehicle-frame specific force, gravity present (m/s²).
 2089:  *   initial_yaw_rad   — ENU course (used only if yaw_valid != 0).
 2090:  *   yaw_valid         — 1 if GNSS heading was reliable, 0 otherwise. */
 2091: static void ahrs_init_from_accel(ahrs_state_t *A,
 2092:                                  const float accel_vehicle[3],
 2093:                                  float initial_yaw_rad,
 2094:                                  int yaw_valid)
 2095: {
 2096:     memset(A, 0, sizeof(*A));
 2097: 
 2098:     float amag = sqrtf(accel_vehicle[0]*accel_vehicle[0] +
 2099:                        accel_vehicle[1]*accel_vehicle[1] +
 2100:                        accel_vehicle[2]*accel_vehicle[2]);
 2101:     if (amag > 1.0f) {
 2102:         // Standard stationary accel→tilt: gravity vector in body frame
 2103:         // is [Rx, Ry, Rz]*g  ≈  accel_vehicle when at rest.
 2104:         A->pitch = atan2f(-accel_vehicle[0],
 2105:                           sqrtf(accel_vehicle[1]*accel_vehicle[1] +
 2106:                                 accel_vehicle[2]*accel_vehicle[2]));
 2107:         A->roll  = atan2f(accel_vehicle[1], accel_vehicle[2]);
 2108:     } else {
 2109:         A->roll  = 0.0f;
 2110:         A->pitch = 0.0f;
 2111:     }
 2112: 
 2113:     if (yaw_valid) {
 2114:         A->yaw     = wrap_pi(initial_yaw_rad);
 2115:         A->yaw_var = (5.0f * DEG2RAD) * (5.0f * DEG2RAD);   // GNSS course σ≈3°
 2116:         A->quality = AHRS_QUALITY_GOOD;
 2117:     } else {
 2118:         A->yaw     = 0.0f;
 2119:         A->yaw_var = (60.0f * DEG2RAD) * (60.0f * DEG2RAD); // unknown
 2120:         A->quality = AHRS_QUALITY_YAW_UNOBSERVED;
 2121:     }
 2122: 
 2123:     A->q = q_from_euler(A->roll, A->pitch, A->yaw);
 2124: 
 2125:     A->roll_var          = (5.0f * DEG2RAD) * (5.0f * DEG2RAD);
 2126:     A->pitch_var         = (5.0f * DEG2RAD) * (5.0f * DEG2RAD);
 2127:     A->accel_norm_lpf    = amag;
 2128:     A->gyro_norm_lpf     = 0.0f;
 2129:     A->roll_pitch_alpha  = AHRS_ROLLPITCH_ALPHA;
 2130:     A->yaw_alpha_gnss    = AHRS_YAW_GNSS_ALPHA;
 2131: 
 2132:     A->initialized       = 1;
 2133:     A->accel_trusted     = 1;
 2134:     A->gnss_yaw_trusted  = yaw_valid;
 2135:     A->last_update_s     = now_sec();
 2136:     A->last_yaw_obs_s    = yaw_valid ? A->last_update_s : 0.0;
 2137: }

 2139: /* Accelerometer trust check.
 2140:  * Trusted iff the body is in quasi-static motion:
 2141:  *     accel norm close to gravity AND gyro low AND no bump active AND
 2142:  *     no large longitudinal acceleration inferred from GNSS speed change. */
 2143: static int ahrs_accel_is_trusted(const float accel_vehicle[3],
 2144:                                  const float gyro_vehicle[3],
 2145:                                  float vehicle_speed_mps)
 2146: {
 2147:     (void)vehicle_speed_mps;
 2148:     float amag = sqrtf(accel_vehicle[0]*accel_vehicle[0] +
 2149:                        accel_vehicle[1]*accel_vehicle[1] +
 2150:                        accel_vehicle[2]*accel_vehicle[2]);
 2151:     if (fabsf(amag - GRAVITY) > AHRS_ACCEL_NORM_TOL) return 0;
 2152:     if (fabsf(gyro_vehicle[0]) > AHRS_ACCEL_TRUST_GYRO_MAX) return 0;
 2153:     if (fabsf(gyro_vehicle[1]) > AHRS_ACCEL_TRUST_GYRO_MAX) return 0;
 2154:     if (fabsf(gyro_vehicle[2]) > AHRS_ACCEL_TRUST_GYRO_MAX) return 0;
 2155:     if (g_imu_quality_degraded) return 0;   // bump detector vetoes accel correction
 2156:     return 1;
 2157: }

 2159: /* GNSS-yaw trust check. */
 2160: static int ahrs_gnss_yaw_is_trusted(int gnss_heading_valid,
 2161:                                     float gnss_heading_rad,
 2162:                                     float gnss_speed_mps,
 2163:                                     float gnss_hdop,
 2164:                                     float current_yaw_rad)
 2165: {
 2166:     if (!gnss_heading_valid) return 0;
 2167:     /* If the most recent fix was rejected by gnss_validate_for_fusion, the
 2168:      * underlying lat/lon/speed/course is not trustworthy — the AHRS must
 2169:      * not pull yaw toward a course derived from invalid GNSS data. */
 2170:     if (!g_gnss_last_fix_validated) return 0;
 2171:     if (gnss_speed_mps < AHRS_GNSS_YAW_SPEED_MIN) return 0;
 2172:     if (gnss_hdop > AHRS_GNSS_YAW_HDOP_MAX) return 0;
 2173:     float innov = wrap_pi(gnss_heading_rad - current_yaw_rad);
 2174:     if (fabsf(innov) > AHRS_YAW_INNOV_MAX) return 0;
 2175:     return 1;
 2176: }

 2178: /* Remove the gravity contribution from accel_vehicle using AHRS attitude.
 2179:  * Output is the linear (vehicle-frame) acceleration excluding gravity.
 2180:  *
 2181:  *   gravity_body = R^T * [0, 0, g]  with  R = R_from_q(A->q)  (body→ENU)
 2182:  *   lin_accel    = accel_vehicle − gravity_body
 2183:  *
 2184:  * For a level vehicle, gravity_body = [0, 0, g] and lin_accel[2] removes g,
 2185:  * lin_accel[0] is forward acceleration, lin_accel[1] is lateral. */
 2186: static void ahrs_remove_gravity(const ahrs_state_t *A,
 2187:                                 const float accel_vehicle[3],
 2188:                                 float lin_accel_vehicle[3])
 2189: {
 2190:     float R[9]; R_from_q(A->q, R);
 2191:     /* R^T column 2 = R row 2 = [R[6], R[7], R[8]] — gravity expressed in body. */
 2192:     float gx_b = R[6] * GRAVITY;
 2193:     float gy_b = R[7] * GRAVITY;
 2194:     float gz_b = R[8] * GRAVITY;
 2195:     lin_accel_vehicle[0] = accel_vehicle[0] - gx_b;
 2196:     lin_accel_vehicle[1] = accel_vehicle[1] - gy_b;
 2197:     lin_accel_vehicle[2] = accel_vehicle[2] - gz_b;
 2198: }

 2200: /* Central AHRS update.
 2201:  * Must be called every IMU step AFTER imu_preprocess_sample() and BEFORE
 2202:  * ins2d_predict(). Updates A->q and the derived euler outputs.            */
 2203: static void ahrs_update(ahrs_state_t *A,
 2204:                         const float accel_vehicle[3],
 2205:                         const float gyro_vehicle[3],
 2206:                         float dt,
 2207:                         int   gnss_heading_valid,
 2208:                         float gnss_heading_rad,
 2209:                         float gnss_speed_mps,
 2210:                         float gnss_hdop)
 2211: {
 2212: #if !ENABLE_AHRS_FILTER
 2213:     (void)A; (void)accel_vehicle; (void)gyro_vehicle; (void)dt;
 2214:     (void)gnss_heading_valid; (void)gnss_heading_rad; (void)gnss_speed_mps; (void)gnss_hdop;
 2215:     return;
 2216: #else
 2217:     if (!A->initialized) return;
 2218:     double now_s = now_sec();
 2219:     A->last_update_s = now_s;
 2220: 
 2221:     /* Diagnostic norms (LPF over a few seconds for stability) */
 2222:     const float dn_alpha = 0.05f;
 2223:     float amag = sqrtf(accel_vehicle[0]*accel_vehicle[0] +
 2224:                        accel_vehicle[1]*accel_vehicle[1] +
 2225:                        accel_vehicle[2]*accel_vehicle[2]);
 2226:     float gmag = sqrtf(gyro_vehicle [0]*gyro_vehicle [0] +
 2227:                        gyro_vehicle [1]*gyro_vehicle [1] +
 2228:                        gyro_vehicle [2]*gyro_vehicle [2]);
 2229:     A->accel_norm_lpf = (1.0f - dn_alpha)*A->accel_norm_lpf + dn_alpha*amag;
 2230:     A->gyro_norm_lpf  = (1.0f - dn_alpha)*A->gyro_norm_lpf  + dn_alpha*gmag;
 2231: 
 2232:     /* 1) Predict: integrate gyro into the quaternion.
 2233:      *    The gyro vector here is already vehicle-frame, bias-removed by the
 2234:      *    preprocessor + runtime cal — we don't subtract another bias here. */
 2235:     vec3f w = v3(gyro_vehicle[0], gyro_vehicle[1], gyro_vehicle[2]);
 2236:     quatf dq = q_from_small_angle(v3_scale(w, dt));
 2237:     A->q = q_normalize(q_mul(A->q, dq));
 2238: 
 2239:     /* Process-noise-like growth of uncertainty per step */
 2240:     float sigma_w_step2 = (0.5f * DEG2RAD)*(0.5f * DEG2RAD) * dt;   // (0.5°/s/√Hz)²·dt
 2241:     A->roll_var  += sigma_w_step2;
 2242:     A->pitch_var += sigma_w_step2;
 2243:     A->yaw_var   += sigma_w_step2;
 2244: 
 2245:     /* 2) Accelerometer roll/pitch correction (gated) */
 2246: #if ENABLE_AHRS_ACCEL_CORR
 2247:     int acc_ok = ahrs_accel_is_trusted(accel_vehicle, gyro_vehicle, gnss_speed_mps);
 2248:     A->accel_trusted = acc_ok;
 2249:     if (acc_ok) {
 2250:         /* Tilt from gravity vector in vehicle frame */
 2251:         float roll_acc  = atan2f(accel_vehicle[1], accel_vehicle[2]);
 2252:         float pitch_acc = atan2f(-accel_vehicle[0],
 2253:                                  sqrtf(accel_vehicle[1]*accel_vehicle[1] +
 2254:                                        accel_vehicle[2]*accel_vehicle[2]));
 2255:         /* Read current euler from q, then blend roll/pitch only, keep yaw */
 2256:         float r_cur, p_cur, y_cur;
 2257:         q_to_euler_zyx(A->q, &r_cur, &p_cur, &y_cur);
 2258:         float r_new = (1.0f - A->roll_pitch_alpha)*r_cur + A->roll_pitch_alpha*roll_acc;
 2259:         float p_new = (1.0f - A->roll_pitch_alpha)*p_cur + A->roll_pitch_alpha*pitch_acc;
 2260:         A->q   = q_from_euler(r_new, p_new, y_cur);
 2261:         A->roll  = r_new;
 2262:         A->pitch = p_new;
 2263:         A->yaw   = y_cur;
 2264:         A->roll_var  *= (1.0f - A->roll_pitch_alpha);
 2265:         A->pitch_var *= (1.0f - A->roll_pitch_alpha);
 2266:         A->accept_accel_count++;
 2267: #if ENABLE_AHRS_DEBUG_LOGS
 2268:         cal_log("[AHRS_ACCEL_ACCEPT] roll_acc=%.2f pitch_acc=%.2f",
 2269:                 roll_acc/DEG2RAD, pitch_acc/DEG2RAD);
 2270: #endif
 2271:     } else {
 2272:         A->reject_accel_count++;
 2273: #if ENABLE_AHRS_DEBUG_LOGS
 2274:         cal_log("[AHRS_ACCEL_REJECT] reason=%s accel_norm=%.2f gyro_norm=%.2f",
 2275:                 g_imu_quality_degraded ? "bump" : "dynamic",
 2276:                 amag, gmag/DEG2RAD);
 2277: #endif
 2278:     }
 2279: #endif /* ENABLE_AHRS_ACCEL_CORR */
 2280: 
 2281:     /* 3) GNSS yaw / course correction (gated) */
 2282: #if ENABLE_AHRS_GNSS_YAW
 2283:     /* Need a current yaw estimate that reflects step (1) and (2) */
 2284:     float r_cur2, p_cur2, y_cur2;
 2285:     q_to_euler_zyx(A->q, &r_cur2, &p_cur2, &y_cur2);
 2286:     int yaw_ok = ahrs_gnss_yaw_is_trusted(gnss_heading_valid, gnss_heading_rad,
 2287:                                           gnss_speed_mps, gnss_hdop, y_cur2);
 2288:     A->gnss_yaw_trusted = yaw_ok;
 2289:     if (yaw_ok) {
 2290:         float innov = wrap_pi(gnss_heading_rad - y_cur2);
 2291:         float y_new = wrap_pi(y_cur2 + A->yaw_alpha_gnss * innov);
 2292:         A->q = q_from_euler(r_cur2, p_cur2, y_new);
 2293:         A->roll  = r_cur2;
 2294:         A->pitch = p_cur2;
 2295:         A->yaw   = y_new;
 2296:         A->yaw_var *= (1.0f - A->yaw_alpha_gnss);
 2297:         A->gnss_yaw_update_count++;
 2298:         A->last_yaw_obs_s = now_s;
 2299: #if ENABLE_AHRS_DEBUG_LOGS
 2300:         cal_log("[AHRS_GNSS_YAW_UPDATE] old=%.2f gnss=%.2f new=%.2f innov=%.2f",
 2301:                 y_cur2/DEG2RAD, gnss_heading_rad/DEG2RAD, y_new/DEG2RAD, innov/DEG2RAD);
 2302: #endif
 2303:     }
 2304: #endif /* ENABLE_AHRS_GNSS_YAW */
 2305: 
 2306:     /* 4) Refresh Euler outputs from authoritative q */
 2307:     q_to_euler_zyx(A->q, &A->roll, &A->pitch, &A->yaw);
 2308: 
 2309:     /* 5) Quality classification (used by EKF for process-noise adaptation) */
 2310:     if (g_imu_quality_degraded) {
 2311:         A->quality = AHRS_QUALITY_BUMP;
 2312:     } else if (!A->accel_trusted && (now_s - A->last_yaw_obs_s) > 30.0) {
 2313:         A->quality = AHRS_QUALITY_BAD;
 2314:     } else if (!A->accel_trusted) {
 2315:         A->quality = AHRS_QUALITY_ACCEL_REJECTED;
 2316:     } else if ((now_s - A->last_yaw_obs_s) > 60.0) {
 2317:         A->quality = AHRS_QUALITY_YAW_UNOBSERVED;
 2318:     } else {
 2319:         A->quality = AHRS_QUALITY_GOOD;
 2320:     }
 2321: #endif /* ENABLE_AHRS_FILTER */
 2322: }
